\subsection*{Exercice 1 :}
Calculer les limites suivantes :
\begin{enumerate}
    \item $\displaystyle\lim_{x\to 2}\dfrac{2x^3-4x^2-3x+6}{x^2-4}$
    \item $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{x^2+x-1}-x}{\sqrt{x}}$
    \item $\displaystyle\lim_{x\to +\infty}\dfrac{\operatorname{sh}(x)}{e^x}$
    \item $\displaystyle\lim_{x\to +\infty}\sqrt{x^2+x}-x$
    \item $\displaystyle\lim_{x\to +\infty}\dfrac{\cos(x^2+x-1)}{x}$
\end{enumerate}

\subsection*{Exercice 2 :}
Soit $f$ la fonction définie par $f(x)=\dfrac{e^x}{e^x+1}$.
\begin{enumerate}
    \item Déterminer le domaine de définition de $f$.
    \item Déterminer les limites de $f$ en $+\infty$ et en $-\infty$.
    \item En déduire l'existence d'asymptotes horizontales à la courbe représentative de $f$, et donner leurs équations.
\end{enumerate}

\subsection*{Exercice 3 :}
Pour tout réel $x$, on pose $f(x)=\dfrac{e^x-e^{-x}}{e^x+e^{-x}}$.
\begin{enumerate}
    \item Déterminer les limites de $f$ en $+\infty$ et en $-\infty$. Que peut-on en déduire pour la courbe représentative de $f$ ?
    \item Démontrer que pour tout réel $x$, on a : $f'(x)=1-f(x)^2$.
    \item En déduire les variations de $f$ et dresser son tableau de variations.
    \item Montrer que $f$ est impaire.
\end{enumerate}

\subsection*{Exercice 4 :}
Soit $f$ la fonction définie par $f(x)=\dfrac{2x^2+3x+9}{x+1}$.
\begin{enumerate}
    \item Déterminer le domaine de définition de $f$.
    \item Déterminer les limites de $f$ aux bornes de son domaine de définition. En déduire l'existence d'une asymptote verticale.
    \item Étudier les variations de $f$ et dresser son tableau de variations.
    \item Montrer que la droite d'équation $y=2x+1$ est asymptote oblique à la courbe représentative de $f$ en $+\infty$ et en $-\infty$. Étudier la position relative de la courbe et de cette asymptote.
    \item Prouver que le point $(-1,-1)$ est centre de symétrie de la courbe représentative de $f$.
\end{enumerate}
